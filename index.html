async function processRawRows(rawRows) { if (!rawRows || rawRows.length === 0) return false; let lastStudentName = ""; let lastBirthday = ""; let lastRoom = ""; let lastTotal = 0; memoryStudents = await Promise.all(rawRows.map(async row => { let name = (row['學生姓名'] || row['姓名'] || row['姓名一'] || row['團員姓名'] || '').toString().trim(); let bdayRaw = (row['生日'] || row['學生生日'] || row['請填寫學生生日:'] || '').toString().trim(); let room = (row['分房(新)'] || row['分房'] || row['房型'] || '').toString().trim(); let totalStr = (row['總費用'] || row['費用(個人)'] || row['費用'] || '').toString().trim(); room = cleanRoomName(room); if (name) lastStudentName = name; else name = lastStudentName; if (bdayRaw) lastBirthday = bdayRaw; else bdayRaw = lastBirthday; if (room) lastRoom = room; else room = lastRoom; let totalCost = Number(totalStr.toString().replace(/[^0-9]/g, '')); if (totalCost > 0) lastTotal = totalCost; else totalCost = lastTotal; const normBday = normalizeDateString(bdayRaw); const nameHash = name ? await sha256(name) : ''; const bdayHash = normBday ? await sha256(normBday) : ''; // 🔴【核心修復】第一期金額 (Col I)：嚴格排除含「確認、時間、備註、方式」的欄位 let deposit1 = 0; const allKeys = Object.keys(row); // 優先尋找名稱同時含有「第一期」與「金額」且不含「確認/時間/方式/備註/回報」的純金額欄位 const exactAmtKey1 = allKeys.find(k => { const isStage1 = k.includes('第一期') || k.includes('訂金'); const isAmt = k.includes('金額') || k.includes('款'); const isNotMeta = !k.includes('確認') && !k.includes('時間') && !k.includes('方式') && !k.includes('備註') && !k.includes('回報') && !k.includes('日期'); return isStage1 && isAmt && isNotMeta; }); if (exactAmtKey1 && row[exactAmtKey1]) { // 確保只提取純數字（避免把含有時間戳記的欄位拿來算） const valStr = row[exactAmtKey1].toString().trim(); // 如果值裡面包含 "確認" 或 ":" 就跳過，避免誤抓 J 欄 if (!valStr.includes('確認') && !valStr.includes(':')) { deposit1 = Number(valStr.replace(/[^0-9]/g, '')) || 0; } } // 備援：若上面沒找到，直接以數字索引或明確標頭名稱回退 if (deposit1 === 0) { const directKey = row['第一期廠商收款金額'] || row['第一期金額'] || row['第一期已繳金額'] || row['訂金金額']; if (directKey && !directKey.toString().includes('確認')) { deposit1 = Number(directKey.toString().replace(/[^0-9]/g, '')) || 0; } } const confirmColJ = (row['第一期金額家長確認'] || row['家長確認'] || row['第一期確認'] || '家長已確認').toString().trim(); // 🔴 第二期家長填寫的回報資料 (Col K~N & Col R) const payMethod2 = (row['第二期家長回報匯款方式'] || '').toString().trim(); const payDate2 = (row['第二期家長回報匯款日期'] || '').toString().trim(); const payLast5_2 = (row['第二期家長回報帳號後五碼'] || '').toString().trim(); const payAmt2 = (row['第二期家長回報匯款金額'] || '').toString().trim(); const confirmColR = (row['第二期金額家長確認'] || '尚未確認').toString().trim(); // 第二期廠商已收款金額 (Col Q) let deposit2 = 0; const exactAmtKey2 = allKeys.find(k => { const isStage2 = k.includes('第二期'); const isAmt = k.includes('金額') || k.includes('款'); const isVendor = k.includes('廠商') || k.includes('收款'); const isNotMeta = !k.includes('家長') && !k.includes('確認') && !k.includes('時間'); return isStage2 && isAmt && isVendor && isNotMeta; }); if (exactAmtKey2 && row[exactAmtKey2]) { deposit2 = Number(row[exactAmtKey2].toString().replace(/[^0-9]/g, '')) || 0; } return { rawName: name, rawBdayNorm: normBday, nameHash: nameHash, birthdayHash: bdayHash, room: room || '未設定', total: totalCost, deposit1: deposit1, confirmColJ: confirmColJ, deposit2: deposit2, payMethod2: payMethod2, payDate2: payDate2, payLast5_2: payLast5_2, payAmt2: payAmt2, confirmColR: confirmColR }; })); memoryStudents = memoryStudents.filter(s => s.rawName.length > 0 && !s.rawName.includes('已成功從 GAS 雲端同步 ${memoryStudents.length} 筆對帳資料`; if (isAdminMode) renderAdminDashboard(); return true; }
